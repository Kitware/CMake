macro(add_sycl_test_macro name)
  if(_isMultiConfig)
    list(APPEND ${name}_CTEST_OPTIONS -C $<CONFIGURATION>)
  endif()
  add_test_macro("${name}" ${ARGN})
  set_property(TEST "${name}" APPEND PROPERTY LABELS "SYCL")
endmacro()

add_sycl_test_macro(SyclOnly.CompileFlags SyclOnlyCompileFlags)
add_sycl_test_macro(SyclOnly.EnableStandard SyclOnlyEnableStandard)
add_sycl_test_macro(SyclOnly.TryCompileTargetStatic SyclOnlyTryCompileTargetStatic)
add_sycl_test_macro(SyclOnly.WithDefs SyclOnlyWithDefs)
add_sycl_test_macro(SyclOnly.CircularLinkLine ${CMAKE_CTEST_COMMAND} -C $<CONFIGURATION> --output-on-failure)
add_sycl_test_macro(SyclOnly.SeparateCompilation ${CMAKE_CTEST_COMMAND} -C $<CONFIGURATION> --output-on-failure)
add_sycl_test_macro(SyclOnly.SeparateCompilationTargetObjects ${CMAKE_CTEST_COMMAND} -C $<CONFIGURATION> --output-on-failure)
add_sycl_test_macro(SyclOnly.ResolveDeviceSymbols ${CMAKE_CTEST_COMMAND} -C $<CONFIGURATION> --output-on-failure)
