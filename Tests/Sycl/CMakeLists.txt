macro(add_sycl_test_macro name)
  if(_isMultiConfig)
    list(APPEND ${name}_CTEST_OPTIONS -C $<CONFIGURATION>)
  endif()
  add_test_macro("${name}" ${ARGN})
  set_property(TEST "${name}" APPEND PROPERTY LABELS "SYCL")
endmacro()

add_sycl_test_macro(Sycl.CXXStandardSetTwice CXXStandardSetTwice)
add_sycl_test_macro(Sycl.ObjectLibrary SyclObjectLibrary)
add_sycl_test_macro(Sycl.MixedStandardLevels1 MixedStandardLevels1)
add_sycl_test_macro(Sycl.MixedStandardLevels2 MixedStandardLevels2)
add_sycl_test_macro(Sycl.MixedStandardLevels3 MixedStandardLevels3)
add_sycl_test_macro(Sycl.MixedStandardLevels4 MixedStandardLevels4)
add_sycl_test_macro(Sycl.MixedStandardLevels5 MixedStandardLevels5)
add_sycl_test_macro(Sycl.NotEnabled SyclNotEnabled)
add_sycl_test_macro(Sycl.SeparableCompCXXOnly SeparableCompCXXOnly)
add_sycl_test_macro(Sycl.ProperLinkFlags ${CMAKE_CTEST_COMMAND} -C $<CONFIGURATION> --output-on-failure)
add_sycl_test_macro(Sycl.Complex ${CMAKE_CTEST_COMMAND} -C $<CONFIGURATION> --output-on-failure)
add_sycl_test_macro(Sycl.WithC SyclWithC)
